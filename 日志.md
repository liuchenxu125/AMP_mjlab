def get_spec() -> mujoco.MjSpec:
    spec = mujoco.MjSpec.from_file(str(MY_ROBOT_XML))

    # 移除 XML 中预定义的 <motor> 元素, 通过 UnitreeActuatorCfg 重新创建。
    for act in list(spec.actuators):
        spec.delete(act)

    return spec

    for n in names:
